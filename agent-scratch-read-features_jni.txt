     1→//! Features JNI (stage F1): deco engine exports + JVM upcall bridge for the
     2→//! 3x3 WorldGenRegion queries (states/heights/biomes/canSurvive).
     3→//!
     4→//! Java protocol (com.planeta.crussty.ChunkGenNative):
     5→//!   decoStates(ctx) -> String[]
     6→//!   decoPlan(ctx) -> String[][]      (steps x names, FeatureSorter order)
     7→//!   decoChunkPlan(ctx, String[] regionBiomes, int[] out) -> int runs
     8→//!       out[2*k] = step, out[2*k+1] = index-in-step (step-major, index-ascending)
     9→//!   decoRunFeature(ctx, long decoSeed, long paperSeed, int originX, originY,
    10→//!       originZ, int step, int gid, DecoLevelBridge bridge, int[] out) -> int ops
    11→//!       out[6*k ..+6] = x, y(abs), z, stateIdx, flags, aux
    12→//!   decoDecoSeed(long levelSeed, int originX, int originZ) -> long
    13→//!
    14→//! DecoLevelBridge upcalls (resolved per decoRunFeature call):
    15→//!   int height(int ty, int x, int z)          — vanilla getHeight semantics
    16→//!   boolean stateIs(int x, int y, int z, String[] names)
    17→//!   boolean fluidIs(int x, int y, int z, String[] names)
    18→//!   boolean isReplaceable(int x, int y, int z)
    19→//!   boolean isSolid(int x, int y, int z)
    20→//!   boolean hasSturdyFace(int x, int y, int z, int dir)
    21→//!   boolean wouldSurvive(String state, int x, int y, int z)
    22→//!   boolean biomeHasFeature(int x, int y, int z, String placed)
    23→//!   boolean outsideBuildHeight(int y)
    24→//!   String stateFull(int x, int y, int z)      — full "block[props]" read
    25→//!   boolean isSolidRender(int x, int y, int z) — PlaceOnGround decorator
    26→//!
    27→//! NOTE: the bridge methods must NOT call back into ChunkGenNative natives —
    28→//! SESSIONS is held for the duration of decoRunFeature (std Mutex is not
    29→//! reentrant; see surface_assign_ids deadlock history).

    37→// run-phase split profiler (CRUS_RUN_PROF=1): where does the decoRunFeature
    38→// wall go — native compute vs the snapshot loader JNI (sections) vs the other
    39→// bridge upcalls. Accumulated across calls; printed + reset at decoChunkBegin
    40→// every PROF_EVERY chunks.
    41→// ---------------------------------------------------------------------------
    42→static PROF_LOADER_NS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    43→static PROF_LOADER_N: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    44→static PROF_PARSE_NS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    45→static PROF_NATIVE_NS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    46→static PROF_RUNS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    47→static PROF_STATE_FULL_NS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    48→static PROF_STATE_FULL_N: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    49→static PROF_SOLIDR_NS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    50→static PROF_SOLIDR_N: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    51→static PROF_CHUNKS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    52→const PROF_EVERY: u64 = 16;
    53→
    54→static PROF_EPOCH: std::sync::OnceLock<std::time::Instant> = std::sync::OnceLock::new();
    55→
    56→#[inline]
    57→fn prof_now() -> u64 {
    58→    PROF_EPOCH.get_or_init(std::time::Instant::now).elapsed().as_nanos() as u64
    59→}
    60→
    61→fn prof_chunk_tick() {
    62→    let n = PROF_CHUNKS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
    63→    if n % PROF_EVERY != PROF_EVERY - 1 {
    64→        return;
    65→    }
    66→    let l = PROF_LOADER_NS.swap(0, std::sync::atomic::Ordering::Relaxed);
    67→    let ln = PROF_LOADER_N.swap(0, std::sync::atomic::Ordering::Relaxed);
    68→    let p = PROF_PARSE_NS.swap(0, std::sync::atomic::Ordering::Relaxed);
    69→    let nat = PROF_NATIVE_NS.swap(0, std::sync::atomic::Ordering::Relaxed);
    70→    let runs = PROF_RUNS.swap(0, std::sync::atomic::Ordering::Relaxed);
    71→    let sf = PROF_STATE_FULL_NS.swap(0, std::sync::atomic::Ordering::Relaxed);
    72→    let sfn = PROF_STATE_FULL_N.swap(0, std::sync::atomic::Ordering::Relaxed);
    73→    let sr = PROF_SOLIDR_NS.swap(0, std::sync::atomic::Ordering::Relaxed);
    74→    let srn = PROF_SOLIDR_N.swap(0, std::sync::atomic::Ordering::Relaxed);
    75→    crate::ffi::log_warn(&format!(
    76→        "[run-prof] chunks=16 runs={} native={:.3}ms loader={:.3}ms/{}sec (jni={:.3} parse={:.3}) state_full={:.3}ms/{} solid_render={:.3}ms/{}",
    77→        runs,
    78→        nat as f64 / 16.0 / 1e6,
    79→        l as f64 / 16.0 / 1e6,
    80→        ln / 16,
    81→        l.saturating_sub(p) as f64 / 16.0 / 1e6,
    82→        p as f64 / 16.0 / 1e6,
    83→        sf as f64 / 16.0 / 1e6,
    84→        sfn / 16,
    85→        sr as f64 / 16.0 / 1e6,
    86→        srn / 16,
    87→    ));
    88→    // ore internals: where the place_ore work goes (exact call counts,
    89→    // cont-118 — the old 1/8 Instant::now sampler distorted x~6 and was
    90→    // JIT/CPU-dependent; wall attribution stays with [run-prof] native=)
    91→    let ore = crate::features_place::ore_prof::drain();
    92→    crate::ffi::log_warn(&format!(
    93→        "[ore-prof] tag={} nodes={} hm={} air={} (calls/window, exact)",
    94→        ore[0], ore[1], ore[2], ore[3],
    95→    ));
    96→    // top-10 features by accumulated native wall (window only)
    97→    let mut top: Vec<(String, u64, u64)> = PROF_FEAT
    98→        .lock()
    99→        .unwrap()
   100→        .iter()
   101→        .map(|(k, v)| (k.clone(), v.0, v.1))
   102→        .collect();
   103→    top.sort_by(|a, b| b.1.cmp(&a.1));
   104→    let mut s = String::from("[run-prof-top]");
   105→    for (i, (name, ns, cnt)) in top.iter().take(10).enumerate() {
   106→        if i > 0 {
   107→            s.push(',');
   108→        }
   109→        s.push_str(&format!(" {}={:.1}us/{}", trim_name(name), *ns as f64 / 16.0 / 1e3, cnt / 16));
   110→    }
   111→    crate::ffi::log_warn(&s);
   112→    PROF_FEAT.lock().unwrap().clear();
   113→}

   115→fn trim_name(n: &str) -> String {
   116→    let short = n.rsplit(':').next().unwrap_or(n);
   117→    short.chars().take(24).collect()
   118→}

   128→// ---------------------------------------------------------------------------
   129→// JVM upcall bridge (DecoLevel impl over the real WorldGenRegion)
   130→// ---------------------------------------------------------------------------
   131→
   132→// cont-135 (Q1, deco-wrapper audit): JniLevel::new resolved 19 method IDs +
   133→// ~19 CString allocs + one std::env::var on EVERY decoRunFeature (~73/chunk)
   134→// and oreRunConfigured (~250/chunk on the ore leg). The bridge class is
   135→// process-stable (one plugin, one classloader, never unloaded), so the table
   136→// is built ONCE into a global ref and reused. jmethodID/jclass are JVM
   137→// process globals — safe to share across gen threads (holder is Send+Sync;
   138→// JNI functions themselves take the caller's env per call).
   139→struct BridgeCache {
   140→    class: jni_sys::jclass, // global ref (kept alive for the mids)
   141→    mids: BridgeMids,
   142→}
   143→unsafe impl Send for BridgeCache {}
   144→unsafe impl Sync for BridgeCache {}
   145→static BRIDGE_CACHE: std::sync::OnceLock<Option<BridgeCache>> = std::sync::OnceLock::new();
   146→static NOUPCALL_ON: std::sync::OnceLock<bool> = std::sync::OnceLock::new();
   147→
   148→#[derive(Clone, Copy)]
   149→struct BridgeMids {
   150→    is_air: jni_sys::jmethodID,
   151→    tag_is: jni_sys::jmethodID,
   152→    state_name: jni_sys::jmethodID,
   153→    name_is_solid: jni_sys::jmethodID,
   154→    height: jni_sys::jmethodID,
   155→    state_is: jni_sys::jmethodID,
   156→    fluid_is: jni_sys::jmethodID,
   157→    is_replaceable: jni_sys::jmethodID,
   158→    is_solid: jni_sys::jmethodID,
   159→    has_sturdy_face: jni_sys::jmethodID,
   160→    would_survive: jni_sys::jmethodID,
   161→    biome_has_feature: jni_sys::jmethodID,
   162→    outside_build_height: jni_sys::jmethodID,
   163→    state_full: jni_sys::jmethodID,
   164→    is_solid_render: jni_sys::jmethodID,
   165→    deco_snap_section: jni_sys::jmethodID,
   166→    deco_snap_section_range: jni_sys::jmethodID,
   167→    deco_heightmaps: jni_sys::jmethodID,
   168→    /// SOFT resolve (not in the ok[] gate): an older plugin jar without
   169→    /// tagMembers only degrades to the per-cell tagIs bridge (None), the
   170→    /// hard-resolved mids above still gate JniLevel::new.
   171→    tag_members: jni_sys::jmethodID,

   173→    /// (native FreezeTopLayerFeature). An older jar without freezeChecks

   175→    /// consumed — safe), every other feature keeps running.
   176→    freeze_checks: jni_sys::jmethodID,
   177→}

   179→pub struct JniLevel<'a> {
   180→    env: *mut jni_sys::JNIEnv,
   181→    obj: jobject,
   182→    mids: BridgeMids,
   183→    /// uid -> placed name (BiomeFilter upcalls carry names)
   184→    uid_names: &'a [String],
   185→    min_y: i32,
   186→    max_y: i32,
   187→    trace: std::cell::Cell<u32>,
   188→    /// CRUS_DECO_NOUPCALL=1: dummy every upcall (height->min_y, bools->false)
   189→    /// — measures the native-only floor of decoRunFeature. Output garbage,
   190→    /// timings honest.
   191→    no_upcall: bool,
   192→}
   193→
   194→impl<'a> JniLevel<'a> {
   195→    fn trace_tick(&self) -> bool {
   196→        let t = self.trace.get();
   197→        self.trace.set(t + 1);
   198→        t < 5
   199→    }
   200→    #[inline]
   201→    fn noup(&self) -> bool {
   202→        self.no_upcall
   203→    }
   204→}
   205→
   206→impl<'a> JniLevel<'a> {
   207→    pub unsafe fn new(
   208→        env: *mut jni_sys::JNIEnv,
   209→        obj: jobject,
   210→        uid_names: &'a [String],
   211→        min_y: i32,
   212→        max_y: i32,
   213→    ) -> Option<Self> {
   214→        let tbl = *env;
   215→        let no_upcall = *NOUPCALL_ON.get_or_init(|| {
   216→            std::env::var("CRUS_DECO_NOUPCALL").ok().as_deref() == Some("1")
   217→        });
   218→        // Fast path: table built on an earlier call — no JNI resolves, one
   219→        // IsSameObject identity guard (plugin reload swaps classloaders:
   220→        // same method names, DIFFERENT Class objects — stale jmethodIDs
   221→        // would be UB; on mismatch the table is rebuilt and the stale
   222→        // global ref dropped so the old classloader can be collected).
   223→        let bc = 'build: {
   224→            if let Some(Some(bc)) = BRIDGE_CACHE.get() {
   225→                let cur = ((*tbl).v1_1.GetObjectClass)(env, obj);
   226→                let same = !cur.is_null()
   227→                    && ((*tbl).v1_1.IsSameObject)(
   228→                        env,
   229→                        cur as jni_sys::jobject,
   230→                        bc.class as jni_sys::jobject,
   231→                    ) == jni_sys::JNI_TRUE;
   232→                if !cur.is_null() {
   233→                    ((*tbl).v1_1.DeleteLocalRef)(env, cur as jni_sys::jobject);
   234→                }
   235→                if same {
   236→                    break 'build bc;
   237→                }
   238→                ((*tbl).v1_1.DeleteGlobalRef)(env, bc.class as jni_sys::jobject);
   239→            }
   240→            let built: Option<BridgeCache> = unsafe {
   241→                    let class = ((*tbl).v1_1.GetObjectClass)(env, obj);
   242→                    if class.is_null() {
   243→                        None
   244→                    } else {
   245→                        let gref = ((*tbl).v1_1.NewGlobalRef)(env, class);
   246→                        ((*tbl).v1_1.DeleteLocalRef)(env, class);
   247→                        let mid = |name: &str, sig: &str| -> jni_sys::jmethodID {
   248→                            let n = std::ffi::CString::new(name).unwrap();
   249→                            let s = std::ffi::CString::new(sig).unwrap();
   250→                            ((*tbl).v1_1.GetMethodID)(env, gref, n.as_ptr(), s.as_ptr())
   251→                        };
   252→                        let mids = BridgeMids {
   253→                            is_air: mid("isAir", "(III)Z"),
   254→                            tag_is: mid("tagIs", "(IIILjava/lang/String;)Z"),
   255→                            state_name: mid("stateName", "(III)Ljava/lang/String;"),
   256→                            name_is_solid: mid("nameIsSolid", "(Ljava/lang/String;)Z"),
   257→                            height: mid("height", "(III)I"),
   258→                            state_is: mid("stateIs", "(III[Ljava/lang/String;)Z"),
   259→                            fluid_is: mid("fluidIs", "(III[Ljava/lang/String;)Z"),
   260→                            is_replaceable: mid("isReplaceable", "(III)Z"),
   261→                            is_solid: mid("isSolid", "(III)Z"),
   262→                            has_sturdy_face: mid("hasSturdyFace", "(IIII)Z"),
   263→                            would_survive: mid("wouldSurvive", "(Ljava/lang/String;III)Z"),
   264→                            biome_has_feature: mid("biomeHasFeature", "(IIILjava/lang/String;)Z"),
   265→                            outside_build_height: mid("outsideBuildHeight", "(I)Z"),
   266→                            state_full: mid("stateFull", "(III)Ljava/lang/String;"),
   267→                            is_solid_render: mid("isSolidRender", "(III)Z"),
   268→                            deco_snap_section: mid("decoSnapSection", "(III)[B"),
   269→                            deco_snap_section_range: mid("decoSnapSectionRange", "(IIII)[B"),
   270→                            deco_heightmaps: mid("decoHeightmaps", "(II)[I"),
   271→                            tag_members: mid("tagMembers", "(Ljava/lang/String;)[Ljava/lang/String;"),
   272→                            freeze_checks: mid("freezeChecks", "(II)I"),
   273→                        };
   274→                        let ok = [
   275→                            mids.is_air,
   276→                            mids.tag_is,
   277→                            mids.state_name,
   278→                            mids.name_is_solid,
   279→                            mids.height,
   280→                            mids.state_is,
   281→                            mids.fluid_is,
   282→                            mids.is_replaceable,
   283→                            mids.is_solid,
   284→                            mids.has_sturdy_face,
   285→                            mids.would_survive,
   286→                            mids.biome_has_feature,
   287→                            mids.outside_build_height,
   288→                            mids.state_full,
   289→                            mids.is_solid_render,
   290→                            mids.deco_snap_section,
   291→                            mids.deco_snap_section_range,
   292→                            mids.deco_heightmaps,
   293→                        ]
   294→                        .iter()
   295→                        .all(|m| !m.is_null());
   296→                        if ok && !gref.is_null() {
   297→                            Some(BridgeCache { class: gref, mids })
   298→                        } else {
   299→                            None
   300→                        }
   301→                    }
   302→                };
   303→                // A None (old jar / broken env) is cached too — the same
   304→                // class resolves the same way on every call, so a per-call
   305→                // retry cannot produce a different answer. (Exception: a
   306→                // plugin RELOAD yields a different Class — caught by the
   307→                // IsSameObject guard above, which forces a rebuild.)
   308→                let _ = BRIDGE_CACHE.set(built);
   309→                match BRIDGE_CACHE.get() {
   310→                    Some(Some(bc)) => bc,
   311→                    _ => return None,
   312→                }
   313→        };
   314→        let mids = bc.mids;
   315→        Some(JniLevel {
   316→            env,
   317→            obj,
   318→            mids,
   319→            uid_names,
   320→            min_y,
   321→            max_y,
   322→            trace: std::cell::Cell::new(0),
   323→            no_upcall,
   324→        })
   325→    }

   327→    #[inline]
   328→    fn z_call(&self, mid: jni_sys::jmethodID, args: &[jni_sys::jvalue]) -> bool {
   329→        if self.noup() {
   330→            return false;
   331→        }
   332→        let tbl = unsafe { *self.env };
   333→        unsafe {
   334→            // A-variant takes a jvalue* — the plain Call*Method is C-varargs
   335→            // and a Rust slice would be read as a single jvalue (JVM crash).
   336→            let r = ((*tbl).v1_1.CallBooleanMethodA)(self.env, self.obj, mid, args.as_ptr());
   337→            if ((*tbl).v1_2.ExceptionCheck)(self.env) {
   338→                ((*tbl).v1_1.ExceptionClear)(self.env);
   339→                return false;
   340→            }
   341→            r
   342→        }
   343→    }

   345→    /// One section snapshot from the bridge (DecoDriver.decoSnapSection).
   346→    /// cont-135 (Q3, deco-wrapper audit): Java now returns its REUSABLE
   347→    /// serialize buffer (GetArrayLength = capacity >= payload; parse_section
   348→    /// is strictly sequential and tolerates the trailing bytes), so both the
   349→    /// Java-side Arrays.copyOf and the native-side to_vec() copy are gone.
   350→    fn snap_section(&self, cx: i32, cz: i32, sec_y: i32) -> Option<crate::features_snap::SnapSection> {
   351→        if self.noup() {
   352→            return None;
   353→        }
   354→        let tbl = unsafe { *self.env };
   355→        unsafe {
   356→            let ivals = [
   357→                Self::val_i(cx),
   358→                Self::val_i(cz),
   359→                Self::val_i(sec_y),
   360→            ];
   361→            let r = ((*tbl).v1_1.CallObjectMethodA)(
   362→                self.env,
   363→                self.obj,
   364→                self.mids.deco_snap_section,
   365→                ivals.as_ptr(),
   366→            ) as jni_sys::jbyteArray;
   367→            if ((*tbl).v1_2.ExceptionCheck)(self.env) {
   368→                ((*tbl).v1_1.ExceptionClear)(self.env);
   369→                return None;
   370→            }
   371→            if r.is_null() {
   372→                return None;
   373→            }
   374→            let len = ((*tbl).v1_1.GetArrayLength)(self.env, r as jni_sys::jarray) as usize;
   375→            if len == 0 {
   376→                ((*tbl).v1_1.DeleteLocalRef)(self.env, r as jni_sys::jobject);
   377→                return None;
   378→            }
   379→            let p = ((*tbl).v1_1.GetByteArrayElements)(self.env, r, std::ptr::null_mut());
   380→            if p.is_null() {
   381→                ((*tbl).v1_1.DeleteLocalRef)(self.env, r as jni_sys::jobject);
   382→                return None;
   383→            }
   384→            let parsed = crate::features_snap::parse_section(std::slice::from_raw_parts(
   385→                p as *const u8,
   386→                len,
   387→            ));
   388→            ((*tbl).v1_2.ReleaseByteArrayElements)(
   389→                self.env,
   390→                r,
   391→                p,
   392→                jni_sys::JNI_ABORT,
   393→            );
   394→            ((*tbl).v1_1.DeleteLocalRef)(self.env, r as jni_sys::jobject);
   395→            parsed
   396→        }
   397→    }

   399→    /// Batch variant: n consecutive sections of one chunk column, one JNI
   400→    /// crossing (DecoDriver.decoSnapSectionRange; length-prefixed payloads).
   401→    fn snap_section_range(&self, cx: i32, cz: i32, sec_y: i32, n: i32) -> Option<Vec<u8>> {
   402→        if self.noup() {
   403→            return None;
   404→        }
   405→        let tbl = unsafe { *self.env };
   406→        unsafe {
   407→            let ivals = [
   408→                Self::val_i(cx),
   409→                Self::val_i(cz),
   410→                Self::val_i(sec_y),
   411→                Self::val_i(n),
   412→            ];
   413→            let r = ((*tbl).v1_1.CallObjectMethodA)(
   414→                self.env,
   415→                self.obj,
   416→                self.mids.deco_snap_section_range,
   417→                ivals.as_ptr(),
   418→            ) as jni_sys::jbyteArray;
   419→            if ((*tbl).v1_2.ExceptionCheck)(self.env) {
   420→                ((*tbl).v1_1.ExceptionClear)(self.env);
   421→                return None;
   422→            }
   423→            if r.is_null() {
   424→                return None;
   425→            }
   426→            let len = ((*tbl).v1_1.GetArrayLength)(self.env, r as jni_sys::jarray) as usize;
   427→            if len == 0 {
   428→                ((*tbl).v1_1.DeleteLocalRef)(self.env, r as jni_sys::jobject);
   429→                return None;
   430→            }
   431→            let p = ((*tbl).v1_1.GetByteArrayElements)(self.env, r, std::ptr::null_mut());
   432→            if p.is_null() {
   433→                ((*tbl).v1_1.DeleteLocalRef)(self.env, r as jni_sys::jobject);
   434→                return None;
   435→            }
   436→            let out = std::slice::from_raw_parts(p as *const u8, len).to_vec();
   437→            ((*tbl).v1_2.ReleaseByteArrayElements)(
   438→                self.env,
   439→                r,
   440→                p,
   441→                jni_sys::JNI_ABORT,
   442→            );
   443→            ((*tbl).v1_1.DeleteLocalRef)(self.env, r as jni_sys::jobject);
   444→            Some(out)
   445→        }
   446→    }

   448→    /// WG heightmap handshake: 512 i32 (WS_WG 256 + OF_WG 256, +1
   449→    /// semantics) per region chunk — replaces the per-column height upcalls.
   450→    fn heightmaps(&self, cx: i32, cz: i32) -> Option<Vec<i32>> {
   451→        if self.noup() {
   452→            return None;
   453→        }
   454→        let tbl = unsafe { *self.env };
   455→        unsafe {
   456→            let ivals = [Self::val_i(cx), Self::val_i(cz)];
   457→            let r = ((*tbl).v1_1.CallObjectMethodA)(
   458→                self.env,
   459→                self.obj,
   460→                self.mids.deco_heightmaps,
   461→                ivals.as_ptr(),
   462→            ) as jni_sys::jintArray;
   463→            if ((*tbl).v1_2.ExceptionCheck)(self.env) {
   464→                ((*tbl).v1_1.ExceptionClear)(self.env);
   465→                return None;
   466→            }
   467→            if r.is_null() {
   468→                return None;
   469→            }
   470→            let len = ((*tbl).v1_1.GetArrayLength)(self.env, r as jni_sys::jarray) as usize;
   471→            if len < 512 {
   472→                ((*tbl).v1_1.DeleteLocalRef)(self.env, r as jni_sys::jobject);
   473→                return None;
   474→            }
   475→            let p = ((*tbl).v1_1.GetIntArrayElements)(self.env, r, std::ptr::null_mut());
   476→            if p.is_null() {
   477→                ((*tbl).v1_1.DeleteLocalRef)(self.env, r as jni_sys::jobject);
   478→                return None;
   479→            }
   480→            let out = std::slice::from_raw_parts(p, 512).to_vec();
   481→            ((*tbl).v1_2.ReleaseIntArrayElements)(
   482→                self.env,
   483→                r,
   484→                p,
   485→                jni_sys::JNI_ABORT,
   486→            );
   487→            ((*tbl).v1_1.DeleteLocalRef)(self.env, r as jni_sys::jobject);
   488→            Some(out)
   489→        }
   490→    }

   874→/// decoBiomeFeaturesFull(ctx, String[] biomeNames, String[] joinedRefs) —
   875→/// the FULL pack biome -> placed-refs union map, matching the Java
   876→/// BiomeFilter source exactly (including biomes OUTSIDE possibleBiomes —
   877→/// border positions sample neighbor biomes whose JSONs still list the
   878→/// feature). joinedRefs[i] = "ref1\u{1}ref2..." (SOH-separated) or empty.
   879→#[no_mangle]
   880→pub unsafe extern "system" fn Java_com_planeta_crussty_ChunkGenNative_decoBiomeFeaturesFull(
   881→    env: *mut jni_sys::JNIEnv,
   882→    _class: jni_sys::jclass,
   883→    ctx: jlong,
   884→    names: jobjectArray,
   885→    joined: jobjectArray,
   886→) {
   887→    let tbl = *env;
   888→    let read_arr = |arr: jobjectArray| -> Vec<String> {
   889→        let mut out = Vec::new();
   890→        unsafe {
   891→            let len = ((*tbl).v1_1.GetArrayLength)(env, arr as jni_sys::jarray) as usize;
   892→            for i in 0..len {
   893→                let jo = ((*tbl).v1_1.GetObjectArrayElement)(env, arr, i as jni_sys::jsize);
   894→                if jo.is_null() {
   895→                    out.push(String::new());
   896→                    continue;
   897→                }
   898→                let chars = ((*tbl).v1_1.GetStringUTFChars)(env, jo as jni_sys::jstring, std::ptr::null_mut());
   899→                let s = if chars.is_null() {
   900→                    String::new()
   901→                } else {
   902→                    std::ffi::CStr::from_ptr(chars).to_string_lossy().into_owned()
   903→                };
   904→                if !chars.is_null() {
   905→                    ((*tbl).v1_1.ReleaseStringUTFChars)(env, jo as jni_sys::jstring, chars);
   906→                }
   907→                ((*tbl).v1_1.DeleteLocalRef)(env, jo);
   908→                out.push(s);
   909→            }
   910→        }
   911→        out
   912→    };

   914→    let refs_v = read_arr(joined);
   915→    // GLOBAL (the chunk deco acquires any ctx from the pool — per-ctx storage
   916→    // would leave the other ctxs with an empty map)
   917→    let mut map = crate::features_snap::BIOME_FULL.write().unwrap();
   918→    map.clear();
   919→    for (i, name) in names_v.iter().enumerate() {
   920→        if name.is_empty() {

   922→        }

   924→            .get(i)
   925→            .map(|j| j.split('\u{1}').filter(|s| !s.is_empty()).map(|s| s.to_string()).collect())
   926→            .unwrap_or_default();

   928→    }

  1217→    };

  1223→        level_ref,
  1224→        ov,
  1225→        ov_cols,
  1226→        snap,
  1227→        &loader,
